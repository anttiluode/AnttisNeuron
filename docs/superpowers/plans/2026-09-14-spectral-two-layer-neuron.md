```python
import numpy as np


def _normalize(v: np.ndarray) -> np.ndarray:
    norm = float(np.linalg.norm(v))
    if norm == 0.0:
        raise ValueError("cannot normalize a zero vector")
    return v / norm


def absolute_cosine(a: np.ndarray, b: np.ndarray) -> float:
    return float(abs(np.dot(_normalize(a), _normalize(b))))


def principal_eigenvector(cov: np.ndarray) -> np.ndarray:
    vals, vecs = np.linalg.eigh(np.asarray(cov, dtype=float))
    return _normalize(vecs[:, int(np.argmax(vals))])


def oja_fit(samples: np.ndarray, *, lr: float = 1e-3, epochs: int = 1, seed: int = 0) -> np.ndarray:
    x = np.asarray(samples, dtype=float)
    if x.ndim != 2 or x.shape[0] < 2:
        raise ValueError("samples must have shape (n_samples, n_features)")
    if lr <= 0.0 or epochs < 1:
        raise ValueError("lr must be positive and epochs >= 1")
    rng = np.random.default_rng(seed)
    w = _normalize(rng.normal(size=x.shape[1]))
    centered = x - x.mean(axis=0, keepdims=True)
    for _ in range(epochs):
        for sample in centered:
            y = float(np.dot(w, sample))
            w += lr * y * (sample - y * w)
            norm = np.linalg.norm(w)
            if not np.isfinite(norm) or norm == 0.0:
                raise FloatingPointError("Oja update became non-finite")
            w /= norm
    return w
```

```python
from dataclasses import dataclass, field
import numpy as np


def spectral_radius(a: np.ndarray) -> float:
    return float(np.max(np.abs(np.linalg.eigvals(np.asarray(a, dtype=float)))))


def activation(name: str, x: np.ndarray) -> np.ndarray:
    x = np.asarray(x, dtype=float)
    if name == "identity":
        return x
    if name == "square":
        return x * x
    if name == "nmda":
        positive = np.maximum(x, 0.0)
        return positive * positive / (1.0 + positive)
    raise ValueError(f"unknown activation: {name}")


def decay_ratio(mu_slow: float, mu_fast: float, t: int) -> float:
    if not (0.0 <= mu_slow < mu_fast):
        raise ValueError("require 0 <= mu_slow < mu_fast")
    return float(np.exp((mu_fast - mu_slow) * t))


@dataclass
class LinearBranch:
    A: np.ndarray
    b: np.ndarray
    c: np.ndarray
    state: np.ndarray = field(init=False)

    def __post_init__(self) -> None:
        self.A = np.asarray(self.A, dtype=float)
        self.b = np.asarray(self.b, dtype=float)
        self.c = np.asarray(self.c, dtype=float)
        n = self.A.shape[0]
        if self.A.shape != (n, n) or self.b.shape != (n,) or self.c.shape != (n,):
            raise ValueError("incompatible branch dimensions")
        if spectral_radius(self.A) >= 1.0:
            raise ValueError("branch must be discrete-time stable")
        self.state = np.zeros(n)

    def reset(self) -> None:
        self.state.fill(0.0)

    def step(self, drive: float) -> float:
        self.state = self.A @ self.state + self.b * float(drive)
        return float(self.c @ self.state)
```

```python
import numpy as np
from anttis_neuron.model import collapsed_linear_filter, quadratic_matrix, static_voltage


def test_linear_branches_collapse_to_one_filter():
    rng = np.random.default_rng(0)
    x = rng.normal(size=(32, 5))
    m = rng.normal(size=(4, 5))
    a = rng.normal(size=4)
    bias = 0.3
    direct = static_voltage(x, m, a, bias=bias, nonlinearity="identity")
    collapsed = x @ collapsed_linear_filter(m, a) + bias
    assert np.allclose(direct, collapsed, atol=1e-12)


def test_square_branches_equal_quadratic_form():
    rng = np.random.default_rng(1)
    x = rng.normal(size=(16, 6))
    m = rng.normal(size=(3, 6))
    a = rng.normal(size=3)
    direct = static_voltage(x, m, a, bias=0.0, nonlinearity="square")
    q = quadratic_matrix(m, a)
    quadratic = np.einsum("bi,ij,bj->b", x, q, x)
    assert np.allclose(direct, quadratic, atol=1e-12)
```

```python
import numpy as np
from .branch import activation


def static_features(x, filters, nonlinearity="identity"):
    x = np.asarray(x, dtype=float)
    filters = np.asarray(filters, dtype=float)
    return activation(nonlinearity, x @ filters.T)


def static_voltage(x, filters, output_weights, *, bias=0.0, nonlinearity="identity"):
    phi = static_features(x, filters, nonlinearity)
    return phi @ np.asarray(output_weights, dtype=float) + float(bias)


def collapsed_linear_filter(filters, output_weights):
    return np.asarray(output_weights, dtype=float) @ np.asarray(filters, dtype=float)


def quadratic_matrix(filters, output_weights):
    m = np.asarray(filters, dtype=float)
    a = np.asarray(output_weights, dtype=float)
    return np.einsum("j,ji,jk->ik", a, m, m)
```

Draw standard Gaussian base noise and independently scale the `u` and `v` coordinates so that class 0 has more energy along `u` and class 1 has more energy along `v`, while both classes remain centered at zero.
